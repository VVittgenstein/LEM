cmake_minimum_required(VERSION 3.20)
project(lem_voxel_backend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LEM_RAYLIB_SOURCE "" CACHE PATH "Existing raylib 5.5 source directory")
if(LEM_RAYLIB_SOURCE)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LEM_RAYLIB_SOURCE}" raylib)
else()
    find_package(raylib 5.5 CONFIG REQUIRED)
endif()
find_package(OpenMP QUIET)
add_library(lem_voxel_core STATIC
    core/fastscape_loader.cpp core/voxel_grid.cpp core/voxel_mesher.cpp
    core/palette_manager.cpp core/camera_controller.cpp core/hud.cpp viewer_input.cpp)
target_include_directories(lem_voxel_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lem_voxel_core PUBLIC raylib)
if(MSVC)
    target_compile_options(lem_voxel_core PRIVATE /utf-8)
    target_compile_definitions(lem_voxel_core PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
if(TARGET OpenMP::OpenMP_CXX)
    target_link_libraries(lem_voxel_core PUBLIC OpenMP::OpenMP_CXX)
endif()
add_executable(lem_voxel_backend backend_main.cpp)
target_link_libraries(lem_voxel_backend PRIVATE lem_voxel_core)
set_target_properties(lem_voxel_backend PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
    add_executable(voxel_core_tests ../../../../tests/viewer/native/test_core.cpp)
    target_link_libraries(voxel_core_tests PRIVATE lem_voxel_core)
    add_test(NAME voxel_core COMMAND voxel_core_tests)
endif()
